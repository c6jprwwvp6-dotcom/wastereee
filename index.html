    • Deep sleep se battery mahino chalti hai
    • Data se predict kar sakte hain kaunsa area kab bharega

    ESP32 Sample Code (Arduino)

    #include <WiFi.h>
#include <HTTPClient.h>
const int TRIG = 5, ECHO = 18;
const float BIN_H = 100, DEAD = 5;          // cm

float readCm() {
  digitalWrite(TRIG, LOW);  delayMicroseconds(2);
  digitalWrite(TRIG, HIGH); delayMicroseconds(10);
  digitalWrite(TRIG, LOW);
  long t = pulseIn(ECHO, HIGH, 30000);      // timeout 30 ms
  return t * 0.0343 / 2;                    // sound speed 343 m/s
}
void setup() {
  pinMode(TRIG, OUTPUT); pinMode(ECHO, INPUT);
  WiFi.begin("SSID", "PASSWORD");
  while (WiFi.status() != WL_CONNECTED) delay(500);
}
void loop() {
  float d = readCm();
  int fill = constrain((BIN_H - d) / (BIN_H - DEAD) * 100, 0, 100);
  HTTPClient http;
  http.begin("https://YOUR-API/bins/B1");   // Firebase / Supabase REST
  http.addHeader("Content-Type", "application/json");
  http.POST("{\"distance\":" + String(d) + ",\"fill\":" + String(fill) + "}");
  http.end();
  esp_sleep_enable_timer_wakeup(10ULL * 60 * 1000000);  // 10 min
  esp_deep_sleep_start();
}
